    df_filtrado
    .write
    .option("header", "true")
    .option("delimiter", ",")
    .csv(VOLUME_PATH)
)

    name="cnpj_bronze",
    comment="Dados brutos do CNPJ ingeridos via Auto Loader"
)

    return (
        spark.readStream
            .format("cloudFiles")
            .option("cloudFiles.format", "csv")
            .option("cloudFiles.inferColumnTypes", "true")
            .option("cloudFiles.schemaEvolutionMode", "addNewColumns")
            .option("header", "true")
            .option("delimiter", ",")
            .load("/Volumes/infnet/bronze/raw_csv/cnpj")

            # Metadados técnicos da ingestão
            .withColumn(
                "_ingestion_timestamp",
                current_timestamp()
            )
            .withColumn(
                "_source_file",
                col("_metadata.file_path")
            )
    )

from pyspark import pipelines as dp
from pyspark.sql.functions import expr


# ============================================================
# EXPECTATIONS
# ============================================================

REGRAS_CNPJ = {

    "cnpj_preenchido":
        "cnpj IS NOT NULL",

    "cnpj_14_digitos":
        "length(cnpj) = 14",

    "situacao_valida":
        "situacao_cadastral IN (1, 2, 3, 4, 8)",

    "data_referencia_preenchida":
        "data_referencia IS NOT NULL",

    "cnae_principal_preenchido":
        "cnae_fiscal_principal IS NOT NULL"
}


# Condição para identificar registros que devem
# ir para a quarentena
QUARANTINE_CONDITION = (
    "NOT("
    + " AND ".join(REGRAS_CNPJ.values())
    + ")"
)


# ============================================================
# ETAPA 3 — QUALITY / EXPECTATIONS
# ============================================================

@dp.table(
    name="cnpj_quality",
    comment="CNPJ com regras de qualidade aplicadas"
)
@dp.expect_all(REGRAS_CNPJ)
def cnpj_quality():

    return (
        spark.readStream
            .table("infnet_lakehouse.bronze.cnpj_bronze")

            .withColumn(
                "is_quarantined",
                expr(QUARANTINE_CONDITION)
            )
    )


# ============================================================
# ETAPA 4 — QUARANTINE
# ============================================================

@dp.table(
    name="cnpj_quarantine",
    comment="Registros do CNPJ que falharam nas regras de qualidade"
)
def cnpj_quarantine():

    return (
        spark.readStream
            .table("cnpj_quality")
            .filter("is_quarantined = true")
    )


# ============================================================
# ETAPA 5 — SILVER QUALITY
# ============================================================

@dp.table(
    name="cnpj_silver_quality",
    comment="CNPJ validado após as regras de qualidade",
    partition_cols=["sigla_uf"],
    table_properties={
        "pipelines.autoOptimize.zOrderCols": "cnpj,id_municipio"
    }
)
def cnpj_silver_quality():

    return (
        spark.readStream
            .table("cnpj_quality")
            .filter("is_quarantined = false")
            .drop("is_quarantined")
    )

---- CDC, SCD, PARTICIONAMENTO E Z-ORDER (PYTHON NOTEBOOK - DLT) ----

from pyspark import pipelines as dp
from pyspark.sql.functions import col


# ============================================================
# 1. IDENTIFICA O PRÓXIMO SNAPSHOT
# ============================================================

def proximo_snapshot(latest_snapshot_version):

    df = spark.read.table(
        "cnpj_silver_quality"
    )

    snapshots = (
        df
        .select("data_referencia")
        .distinct()
        .orderBy("data_referencia")
    )

    # Primeiro snapshot
    if latest_snapshot_version is None:

        rows = (
            snapshots
            .limit(1)
            .collect()
        )

    # Próximo snapshot
    else:

        rows = (
            snapshots
            .filter(
                col("data_referencia")
                > latest_snapshot_version
            )
            .limit(1)
            .collect()
        )

    # Não existe novo snapshot
    if not rows:
        return None

    proxima_data = rows[0]["data_referencia"]

    snapshot = (
        df
        .filter(
            col("data_referencia")
            == proxima_data
        )
    )

    return snapshot, proxima_data

@dp.materialized_view(

        )
        .dropDuplicates(
            ["id_municipio"]

@dp.materialized_view(
    name="dim_cnae",

            "cnae_fiscal_principal"

@dp.materialized_view(
    name="fato_estabelecimentos",
    comment="Quantidade de estabelecimentos por município, CNAE e situação"
)
def fato_estabelecimentos():

    return (
        spark.read.table("infnet_lakehouse.silver.cnpj_silver_quality")

        .groupBy(
            "data_referencia",
            "id_municipio",
            "sigla_uf",
            "cnae_fiscal_principal",
            "situacao_cadastral"
        )

        .agg(
            count("cnpj").alias(
                "quantidade_estabelecimentos"
            )
        )
    )

SELECT
    f.data_referencia,
    m.sigla_uf,
    f.cnae_fiscal_principal,
    SUM(f.quantidade_estabelecimentos) AS quantidade_estabelecimentos

FROM infnet.gold.fato_estabelecimentos AS f

LEFT JOIN infnet.gold.dim_municipio AS m
    ON f.id_municipio = m.id_municipio

WHERE f.situacao_cadastral = 2

GROUP BY
    f.data_referencia,
    m.sigla_uf,
    f.cnae_fiscal_principal

ORDER BY
    f.data_referencia DESC,
    quantidade_estabelecimentos DESC;

WITH ultima_data AS (
    SELECT MAX(data_referencia) AS data_referencia
    FROM infnet.gold.fato_estabelecimentos
)

SELECT
    f.data_referencia,
    m.sigla_uf,
    f.cnae_fiscal_principal,
    SUM(f.quantidade_estabelecimentos) AS quantidade_estabelecimentos

FROM infnet.gold.fato_estabelecimentos AS f

LEFT JOIN infnet.gold.dim_municipio AS m
    ON f.id_municipio = m.id_municipio

INNER JOIN ultima_data AS u
    ON f.data_referencia = u.data_referencia

WHERE f.situacao_cadastral = 2

GROUP BY
    f.data_referencia,
    m.sigla_uf,
    f.cnae_fiscal_principal

ORDER BY
    quantidade_estabelecimentos DESC;
