----- COPY TABLE (PYTHON) -----
spark.sql("""
CREATE TABLE infnet.silver.dados
AS
SELECT *
FROM infnet.bronze.dados
WHERE 1 = 0
""")

----- MERGE/UPSERT (PYTHON) -----
from delta.tables import DeltaTable

# --------------------------------------------------
# 1. Tabela de destino
# --------------------------------------------------

destino = DeltaTable.forName(
    spark,
    "infnet.silver.dados"
)


# --------------------------------------------------
# 2. Seleciona somente os registros novos
#    e pega a última versão de cada matrícula
# --------------------------------------------------

origem = spark.sql("""
    SELECT *
    FROM (
        SELECT
            b.*,
            ROW_NUMBER() OVER (
                PARTITION BY matricula
                ORDER BY ingestion_timestamp DESC
            ) AS rn
        FROM infnet.bronze.dados b
        WHERE b.ingestion_timestamp >
              COALESCE(
                  (
                      SELECT MAX(ingestion_timestamp)
                      FROM infnet.silver.dados
                  ),
                  TIMESTAMP '1900-01-01'
              )
    )
    WHERE rn = 1
""")


# --------------------------------------------------
# 3. MERGE Bronze → Silver
# --------------------------------------------------

(
    destino.alias("destino")
    .merge(
        origem.alias("origem"),
        "destino.matricula = origem.matricula"
    )
    .whenMatchedUpdateAll(
        condition="""
            origem.ingestion_timestamp >
            destino.ingestion_timestamp
        """
    )
    .whenNotMatchedInsertAll()
    .execute()
)

----- Z-ORDER (PYTHON) -----
resultado = spark.sql("""
    OPTIMIZE infnet.silver.dados
    ZORDER BY (matricula)
""")

resultado.show(truncate=False)
