---- MONITORAMENTO DE SEGURANÇA - QUEM ACESSOU O QUÊ? (SQL EDITOR) ---
SELECT
    event_time,
    user_identity.email AS usuario,
    action_name,
    request_params
FROM system.access.audit
WHERE event_time >= current_timestamp() - INTERVAL 1 DAY
ORDER BY event_time DESC;

---- MONITORAMENTO DE TABELA (SQL EDITOR) ---

    user_identity.email AS usuario,
    action_name,
    request_params
FROM system.access.audit

  AND to_json(request_params) LIKE '%infnet.bronze.dados%'
ORDER BY event_time DESC;

---- MONITORAMENTO DE CUSTOS (SQL EDITOR) ---
SELECT
    *
FROM system.billing.usage
WHERE usage_date >= current_date() - INTERVAL 7 DAYS;

SELECT
    identity_metadata.run_as,
    SUM(usage_quantity) AS consumo
FROM system.billing.usage
WHERE usage_date >= current_date() - INTERVAL 7 DAYS
GROUP BY identity_metadata.run_as
ORDER BY consumo DESC;
